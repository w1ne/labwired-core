    ESP32 — rank 1, 38% of reported MCU use

    ESP32-C6-DevKitC-1-N8 + BME280

    The exact N8 board is listed in Espressif's current user guide; the dated 2026-11-10 receipt records 178 in stock at DigiKey. The 8 MB profile boots a direct-register smoke that reads BME280 ID 0x60 and raw temperature over I2C0 on GPIO6/GPIO7. This adds board depth on the existing ESP32-C6 model, not new silicon.

    • Wiring, scope, and copy-paste example
    • Dated Espressif, DigiKey, and Adafruit availability receipt
    • Committed local smoke evidence

    The hosted catalog lists the C6 DevKitC as design_only, so this draft links the exact local CLI smoke and does not claim hosted labwired_verify coverage or physical hardware validation.

    #include <Adafruit_BME280.h>
#include <Arduino.h>
#include <Wire.h>

Adafruit_BME280 bme;

void setup() {
  Serial.begin(115200);
  Wire.begin(6, 7, 100000);  // SDA=GPIO6, SCL=GPIO7
  if (!bme.begin(0x77, &Wire)) {
    Serial.println("BME280 not found");
    return;
  }
  Serial.printf("BME280_ID=0x%02X\n", bme.sensorID());
  Serial.printf("Temperature C: %.2f\n", bme.readTemperature());
}

void loop() {}

    ESP8684-DevKitM-1-H4X + UART0 and red LED

    ESP32 remains demand rank 1 (38% of reported MCU use); ESP32-C2 is not ranked separately. Espressif's live guide lists the H4X order code with 4 MB flash and marks the older H4 variant NRND. Mouser showed 35 H4X boards in stock on 2026-11-17. The board smoke sends a UART0 marker and observes GPIO0 transitions on the onboard red LED channel. This adds a working board profile and C2 clock pacing; it does not claim ROM boot, the USB-UART bridge, or pad electrical routing.

    • Board scope, copy-paste ESP-IDF example, and smoke command
    • Live Espressif lifecycle and Mouser availability receipt
    • Committed UART0 and GPIO0 smoke evidence
    #include "driver/gpio.h"
#include "driver/uart.h"
#include "esp_rom_sys.h"
#include <stddef.h>

void app_main(void) {
    const uart_config_t uart_cfg = {
        .baud_rate = 115200,
        .data_bits = UART_DATA_8_BITS,
        .parity = UART_PARITY_DISABLE,
        .stop_bits = UART_STOP_BITS_1,
        .flow_ctrl = UART_HW_FLOWCTRL_DISABLE,
        .source_clk = UART_SCLK_DEFAULT,
    };
    uart_driver_install(UART_NUM_0, 256, 0, 0, NULL, 0);
    uart_param_config(UART_NUM_0, &uart_cfg);
    gpio_reset_pin(GPIO_NUM_0);
    gpio_set_direction(GPIO_NUM_0, GPIO_MODE_OUTPUT);
    static const char marker[] = "C2_H4X_GPIO0_UART_OK\n";
    uart_write_bytes(UART_NUM_0, marker, sizeof(marker) - 1);
    for (;;) {
        gpio_set_level(GPIO_NUM_0, 1);
        esp_rom_delay_us(250000);
        gpio_set_level(GPIO_NUM_0, 0);
        esp_rom_delay_us(250000);
    }
}

    ESP32-S3-DevKitC-1-N16R8 + Adafruit BME280

    ESP32 remains demand rank 1 (38% of reported MCU use). Espressif lists the N16R8 board SKU; the v1.1 guide gives 16 MB flash / 8 MB PSRAM and marks the older N8/N8R2 variants EOL. Adafruit listed the BME280 breakout in stock on 2026-11-13. The profile uses I2C0 on GPIO8/GPIO9 and asserts the sensor ID plus raw temperature registers. This adds board and sensor coverage over the existing S3 model, not new silicon. The hosted Arduino oracle verifies the copy-paste setup path; its ideal-rail assumption excludes power-on and brown-out behavior.

    • Wiring, scope, and copy-paste Arduino sketch
    • Dated Espressif and Adafruit availability receipt
    • Committed local smoke logs
    • Hosted model_verified oracle output
    #include <Arduino.h>
#include <Wire.h>

constexpr uint8_t SDA_PIN = 8;
constexpr uint8_t SCL_PIN = 9;
constexpr uint8_t BME280_ADDR = 0x77;

bool readRegister(uint8_t reg, uint8_t &value) {
  Wire.beginTransmission(BME280_ADDR);
  Wire.write(reg);
  if (Wire.endTransmission(false) != 0 || Wire.requestFrom(BME280_ADDR, 1) != 1) {
    return false;
  }
  value = Wire.read();
  return true;
}

void setup() {
  Serial.begin(115200);
  Wire.begin(SDA_PIN, SCL_PIN, 100000);
  uint8_t id = 0;
  if (readRegister(0xD0, id) && id == 0x60) {
    Serial.printf("BME280_ID=0x%02X\n", id);
  } else {
    Serial.println("BME280_I2C_ERROR");
  }
}

void loop() {}

    Seeed Studio XIAO ESP32C3 + Adafruit BME280

    The XIAO ESP32C3 is a USD 4.90 ESP32-C3 board; Seeed listed it in stock on 2026-11-13. The breakout sits on the board's default Wire pins, D4 (GPIO6) and D5 (GPIO7). The simulator boots the unmodified Arduino build in full and the sketch prints the BME280 ID, the raw temperature and the compensated 25.00 degC on the USB console. This adds board and sensor coverage over the existing C3 model, not new silicon.

    • Wiring, scope, and copy-paste Arduino sketch
    • Dated Seeed availability receipt
    • Committed local smoke logs
    #include <Arduino.h>
#include <Wire.h>

constexpr uint8_t BME280_ADDR = 0x77;

void setup() {
  Serial.begin(115200);
  Wire.begin();  // XIAO ESP32C3: SDA = D4 (GPIO6), SCL = D5 (GPIO7)
  Wire.beginTransmission(BME280_ADDR);
  Wire.write(0xD0);
  uint8_t id = 0;
  if (Wire.endTransmission(false) == 0 && Wire.requestFrom(BME280_ADDR, 1) == 1) {
    id = Wire.read();
  }
  Serial.printf("BME280_ID=0x%02X\n", id);

    Arduino kit boards — rank 3, 37% of main-project kits

    Arduino UNO R4 Minima + BME280

    Arduino lists the UNO R4 Minima as orderable SKU ABX00080, and Renesas lists RA4M1 orderable parts as Active. The Adafruit BME280 STEMMA QT breakout was in stock on 2026-11-17. This adds an IIC1 sensor path to the existing UNO R4 Minima model; RA4M1 itself is not separately ranked in the MCU-use table.

    • Wiring, scope, copy-paste sketch, and smoke instructions
    • Live vendor status receipt
    • Committed local smoke evidence

    The sketch reads the BME280 chip ID over Wire/IIC1 on A4/A5 and reports it over Serial1/SCI2 on D0/D1. USB-C Serial uses USBFS, which remains outside this profile.

    #include <Arduino.h>
#include <Wire.h>

constexpr uint8_t BME280_ADDRESS = 0x77;

void setup() {
  Serial1.begin(115200);  // SCI2 on D0/D1
  Wire.begin();           // IIC1 on A4/A5
  Wire.setClock(100000);

  Wire.beginTransmission(BME280_ADDRESS);
  Wire.write(0xD0);
  if (Wire.endTransmission(false) != 0 ||
      Wire.requestFrom(static_cast<int>(BME280_ADDRESS), 1) != 1) {
    Serial1.println("BME280_I2C_ERROR");
    return;
  }

  Serial1.print("BME280_ID=0x");
  Serial1.println(Wire.read(), HEX);
}

void loop() {}

    Arduino Nano Every + Adafruit BME280

    Arduino lists the Nano Every as orderable SKU ABX00028, and Microchip's live page reported ATMEGA4809 as In Production on 2026-11-19. The Adafruit BME280 breakout was in stock on the same date. The ATmega4809 is not ranked separately; the Nano Every is the in-production Arduino board in the classic Nano form factor. This adds a TWI0 host model to the ATmega4809, so stock Wire runs.

    • Wiring, scope, copy-paste sketch, and smoke instructions
    • Live vendor status receipt
    • Committed local smoke evidence and controls

    The sketch uses the stock Adafruit BME280 library on A4/A5 and prints compensated readings on USB Serial: T=25.00 C, P=1013.25 hPa, RH=50.00 % with the sensor model at its defaults. Timer pad outputs and the TWI0 pads are not modelled. No physical board test was run.

    #include <Arduino.h>
#include <Wire.h>
#include <Adafruit_BME280.h>

Adafruit_BME280 bme;

void setup() {
  pinMode(LED_BUILTIN, OUTPUT);
  Serial.begin(115200);
  if (!bme.begin(0x77)) {
    Serial.println("BME280 not found at 0x77");
    return;
  }
  Serial.print("NANO_EVERY_BME280 ID=0x");
  Serial.println(bme.sensorID(), HEX);
  Serial.print("T=");
  Serial.print(bme.readTemperature(), 2);
  Serial.println(" C");
  Serial.print("P=");
  Serial.print(bme.readPressure() / 100.0F, 2);
  Serial.println(" hPa");
  Serial.print("RH=");
  Serial.print(bme.readHumidity(), 2);
  Serial.println(" %");
  digitalWrite(LED_BUILTIN, HIGH);
}

void loop() {}

    Arduino Nano Every + Adafruit MAX31855

    Same board and the same live checks on 2026-11-19: Nano Every SKU ABX00028 orderable, ATMEGA4809 In Production. The Adafruit MAX31855 breakout was in stock on that date; the MAX31855 lifecycle page at Analog Devices was not readable. This adds an SPI0 host model to the ATmega4809, so stock SPI runs.

    • Wiring, scope, copy-paste sketch, and smoke instructions
    • Live vendor status receipt
    • Committed local smoke evidence and controls

    The sketch uses the stock Adafruit MAX31855 library on D13 / D12 with CS on D10 and prints both junction temperatures on USB Serial: TC=25.00 C, CJ=22.0000 C with the converter model at its defaults. SPI0 Buffer mode and client mode are refused; the SPI clock phase and polarity and timer pad outputs are not modelled. No physical board test was run.

    #include <Arduino.h>
#include <SPI.h>
#include <Adafruit_MAX31855.h>

static const uint8_t kChipSelect = 10;

Adafruit_MAX31855 thermocouple(kChipSelect);

void setup() {
  Serial.begin(115200);
  thermocouple.begin();
  Serial.println("NANO_EVERY_MAX31855");
  double hot = thermocouple.readCelsius();
  if (isnan(hot)) {
    Serial.print("MAX31855 fault 0x");
    Serial.println(thermocouple.readError(), HEX);
    return;
  }
  Serial.print("TC=");
  Serial.print(hot, 2);
  Serial.println(" C");
  Serial.print("CJ=");
  Serial.print(thermocouple.readInternal(), 4);
  Serial.println(" C");
}

    ESP32-C61-DevKitC-1-N8R2

    ESP32 is demand rank 1 (38% of reported MCU use); the survey does not score C61 separately. Espressif lists the exact N8R2 board, and Mouser showed 378 in stock on 2026-11-17. The C61 profile now exercises UART0, GPIO4 output, and I2C0 routed through GPIO6/GPIO22 to a BME280. It reads the BME280 ID (0x60) and a deterministic raw-temperature vector. ROM/bootloader, electrical IO_MUX, PSRAM, Wi-Fi, Bluetooth LE, and the onboard RGB LED remain outside scope.

    • Board scope, wiring, and copy-paste ESP-IDF example
    • Live Espressif, Mouser, Bosch, and Adafruit availability receipt
    • Committed UART/GPIO and BME280 smoke evidence
    #include <stdint.h>

#include "driver/gpio.h"
#include "driver/i2c_master.h"
#include "esp_err.h"
#include "esp_log.h"

#define BME280_ADDRESS 0x77

void app_main(void) {
    i2c_master_bus_handle_t bus;
    const i2c_master_bus_config_t bus_config = {
        .i2c_port = I2C_NUM_0,
        .sda_io_num = GPIO_NUM_6,   // J1 pin 5
        .scl_io_num = GPIO_NUM_22,  // J3 pin 6
        .clk_source = I2C_CLK_SRC_DEFAULT,
        .glitch_ignore_cnt = 7,
    };
    ESP_ERROR_CHECK(i2c_new_master_bus(&bus_config, &bus));

    i2c_master_dev_handle_t sensor;
    const i2c_device_config_t sensor_config = {
        .dev_addr_length = I2C_ADDR_BIT_LEN_7,
        .device_address = BME280_ADDRESS,
        .scl_speed_hz = 100000,
    };
    ESP_ERROR_CHECK(i2c_master_bus_add_device(bus, &sensor_config, &sensor));

    const uint8_t chip_id_register = 0xD0;
    uint8_t chip_id = 0;
    ESP_ERROR_CHECK(i2c_master_transmit_receive(
        sensor, &chip_id_register, 1, &chip_id, 1, 1000));
    if (chip_id != 0x60) {
        ESP_LOGE("bme280", "unexpected chip id: 0x%02x", chip_id);
        return;
    }
    ESP_LOGI("bme280", "BME280_ID=0x%02x", chip_id);
}
