# LabWired - narrowed SiFive FE310 profile for the imported Renode system.
#
# Renode's platforms/cpus/sifive-fe310.repl identifies the RV32IMAC CPU,
# QSPI XIP at 0x20000000, DTIM at 0x80000000, and uart0 at 0x10013000.
# The imported YAML had an ARM label and mistook mask ROM/OTP for flash/RAM.
# Keep the runnable HiFive1 Rev B memory ranges, UART0 and the software GPIO
# path used by the smoke. The CLINT at 0x02000000 (msip, mtimecmp, mtime at
# the 32 768 Hz lfclk) belongs to the core complex: the RV32 core decodes it
# for this chip name (`Rv32Clint::for_chip`), so it is not a peripheral entry.
# The PLIC at 0x0C000000 (52 sources, 3-bit priorities, one M-mode context)
# is decoded the same way (`Rv32Clint::plic_for_chip`); a peripheral `irq:` is
# its PLIC source ID (uart0 = 3, Renode `IRQ -> plic@3`). The GPIO `irq:` is
# pin 0's source: pin n drives source 8 + n (Renode `[0-31] -> plic@[8-39]`).
# UART1, GPIO IO functions, mask ROM and OTP are not claimed.

name: "sifive-fe310"
arch: "riscv"
core: "rv32imac_zicsr_zifencei"
cpu_hz: 62000000

flash: { base: 0x20000000, size: "4MiB" }
ram: { base: 0x80000000, size: "16KiB" }
peripherals:
  - id: "uart0"
    type: "sifive_uart"
    base_address: 0x10013000
    size: "4KB"
    irq: 3
  - id: "gpio"
    type: "sifive_gpio"
    base_address: 0x10012000
    size: "4KB"
